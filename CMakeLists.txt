cmake_minimum_required(VERSION 3.24)
project(KFVR VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive-)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
add_library(KFVRCore STATIC
 src/math/StereoMath.cpp src/math/StereoCulling.cpp src/math/WeaponPoseMath.cpp
 src/math/UiPointerMath.cpp src/math/UiPointerSmoothing.cpp src/math/ArmPoleVectorMath.cpp
 src/interaction/TrackingMath.cpp src/interaction/ComfortCamera.cpp
 src/interaction/ComfortControls.cpp src/interaction/WeaponGrip.cpp
 src/interaction/ActionPolicy.cpp src/interaction/RigPose.cpp
 src/runtime/FrameCoordinator.cpp src/ipc/FrameProtocol.cpp
 src/engine/Ue2Camera.cpp src/gameplay/AmmoMachine.cpp src/gameplay/MeleeGate.cpp src/gameplay/PoseWire.cpp)
target_include_directories(KFVRCore PUBLIC include include/kfvr/interaction include/kfvr/math)
if(BUILD_TESTING)
 foreach(name StereoCulling FrameProtocol PresentationPolicy SharedTextureProtocol CoreMath FrameCoordinator ActionPolicy RigPose UiPointerSmoothing ArmPoleVectorMath AmmoMachine MeleeGate PoseWire Ue2Camera)
  add_executable(${name}Tests tests/${name}Tests.cpp)
  target_link_libraries(${name}Tests PRIVATE KFVRCore)
  add_test(NAME ${name} COMMAND ${name}Tests)
 endforeach()
endif()
if(WIN32)
 add_library(KFVRD3D11 STATIC src/graphics/d3d11/D3D11SharedPair.cpp)
 target_include_directories(KFVRD3D11 PUBLIC include)
 target_compile_definitions(KFVRD3D11 PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
 target_link_libraries(KFVRD3D11 PUBLIC d3d11 dxgi ole32)
 add_library(KFVRFrameChannel STATIC src/platform/windows/FrameChannel.cpp src/ipc/RemoteFrameProvider.cpp)
 target_link_libraries(KFVRFrameChannel PUBLIC KFVRCore ole32)
 target_compile_definitions(KFVRFrameChannel PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
 add_library(KFVRDiagnosticScene STATIC src/xr/DiagnosticScene.cpp)
 target_link_libraries(KFVRDiagnosticScene PUBLIC KFVRCore KFVRD3D11 d3dcompiler)
 add_executable(KFVRBridgeProbe src/platform/windows/D3D11BridgeProbe.cpp)
 target_link_libraries(KFVRBridgeProbe PRIVATE KFVRD3D11)
 add_executable(KFVRIpcProbe src/platform/windows/IpcBridgeProbe.cpp)
 target_link_libraries(KFVRIpcProbe PRIVATE KFVRFrameChannel KFVRDiagnosticScene)
 if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_library(KFVROpenXR STATIC src/xr/OpenXrHost.cpp)
  target_include_directories(KFVROpenXR PRIVATE third_party/openxr-1.1.61/include)
  target_link_libraries(KFVROpenXR PUBLIC KFVRCore KFVRD3D11)
  add_executable(KFVRXrHost src/platform/windows/OpenXrHostMain.cpp)
  target_link_libraries(KFVRXrHost PRIVATE KFVROpenXR KFVRDiagnosticScene KFVRFrameChannel)
  add_executable(KFVRSceneProbe src/platform/windows/SceneProbe.cpp)
  target_link_libraries(KFVRSceneProbe PRIVATE KFVROpenXR KFVRDiagnosticScene)
  if(EXISTS "${CMAKE_SOURCE_DIR}/runtime/openxr/win64/openxr_loader.dll")
   add_custom_command(TARGET KFVRXrHost POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:KFVRXrHost>/runtime/openxr/win64"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/runtime/openxr/win64/openxr_loader.dll" "$<TARGET_FILE_DIR:KFVRXrHost>/runtime/openxr/win64/openxr_loader.dll")
  endif()
 endif()
endif()

if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
 add_library(KFVRMinHook STATIC third_party/minhook-1.3.4/src/buffer.c third_party/minhook-1.3.4/src/hook.c third_party/minhook-1.3.4/src/trampoline.c third_party/minhook-1.3.4/src/hde/hde32.c)
 target_include_directories(KFVRMinHook PUBLIC third_party/minhook-1.3.4/include)

 target_compile_definitions(KFVRNativeProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 add_executable(KFVRNativeTrace src/platform/windows/NativeTraceMain.cpp)
 target_include_directories(KFVRNativeTrace PRIVATE src/native)
 target_compile_definitions(KFVRNativeTrace PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
